# 🛠️ Fonctions utilitaires
def load_module(module_name: str, **kwargs):
    """Charge dynamiquement un module Python avec gestion d'erreurs."""
    module_path = MODULES_DIR / f"{module_name}.py"
    
    if not module_path.exists():
        st.error(f"Module `{module_name}.py` introuvable dans {MODULES_DIR}")
        return False
    
    try:
        spec = importlib.util.spec_from_file_location(module_name, module_path)
        module = importlib.util.module_from_spec(spec)
        
        # Injection des variables dans le module
        for key, value in kwargs.items():
            setattr(module, key, value)
        
        spec.loader.exec_module(module)
        
        # Important : enregistrer dans sys.modules pour pouvoir le récupérer plus tard
        sys.modules[module_name] = module
        
        return True
    except Exception as e:
        st.error(f"Erreur lors du chargement du module {module_name}: {str(e)}")
        logger.error(f"Erreur module {module_name}: {e}")
        return False

def page_nlp():
    st.header("🧠 Traitement Automatique du Langage Économique")
    
    col1, col2 = st.columns([2, 1])
    
    with col1:
        st.markdown("### 💬 Analysez vos requêtes économiques")
        query = st.text_area(
            "Entrez une requête économique :",
            placeholder="Ex: Analysez l'impact de l'inflation sur le pouvoir d'achat...",
            height=150
        )
        
        if st.button("📤 Lancer l'analyse NLP", type="primary"):
            if query.strip():
                with st.spinner("Analyse en cours..."):
                    st.success(f"Requête reçue : **{query}**")
                    
                    # Chargement du module NLP (exécution et retour d'un DataFrame)
                    success = load_module("nlp_handler", query=query)
                    if success:
                        st.info("✅ Module NLP exécuté avec succès")

                        # Récupérer le DataFrame résultat (ex: nlp_handler.result_df)
                        nlp_result_df = getattr(sys.modules.get("nlp_handler"), "result_df", None)

                        if nlp_result_df is not None and isinstance(nlp_result_df, pd.DataFrame):
                            # S'assurer que les colonnes nécessaires existent
                            if 'date' in nlp_result_df.columns and 'valeur' in nlp_result_df.columns:
                                df = nlp_result_df.copy()

                                # Renommer les colonnes comme demandé
                                df = df.rename(columns={'date': 'Datetime', 'valeur': 'inflation_mom'})

                                # Formater la colonne 'Datetime' en date (sans heure)
                                df['Datetime'] = pd.to_datetime(df['Datetime']).dt.date

                                # Afficher le DataFrame complet dans un style pandas classique
                                st.markdown("### Résultat simplifié (Date et inflation_mom)")
                                st.dataframe(df.style.format({"inflation_mom": "{:.2f}"}), use_container_width=True)

                            else:
                                st.warning("Les colonnes 'date' et/ou 'valeur' sont absentes dans les résultats NLP.")
            else:
                st.warning("Veuillez saisir une requête avant de lancer l'analyse.")

    with col2:
        st.markdown("### 📋 Exemples de requêtes")
        st.markdown("""
        - Impact de l'inflation sur l'économie
        - Analyse des taux d'intérêt
        - Évolution du PIB tunisien
        - Politique monétaire BCT
        """)

def page_prediction():
    st.header("📈 Prédiction de l'Inflation")

    st.markdown("""
    Utilisez notre modèle avancé pour prédire l'évolution de l'inflation 
    en Tunisie basé sur les données historiques et les indicateurs économiques.
    """)

    # Chargement du module prediction
    success = load_module("prediction")
    if not success:
        st.error("❌ Le module `prediction` n'a pas pu être chargé.")
        return

    module = sys.modules.get("prediction")
    if module is None:
        st.error("❌ Le module `prediction` n'a pas été importé correctement.")
        return

    # Recherche dynamique des fonctions disponibles dans prediction.py
    algo_functions = {
        attr: getattr(module, attr)
        for attr in dir(module)
        if callable(getattr(module, attr)) and attr.startswith("run_")
    }

    if not algo_functions:
        st.warning("Aucune fonction de prédiction `run_...` trouvée dans `prediction.py`.")
        return

    # Préparation liste des noms d'algos affichables (ex: 'run_gradient_boosting' -> 'Gradient Boosting')
    def pretty_name(func_name):
        name = func_name.replace("run_", "").replace("_", " ").title()
        return name

    algo_names = {pretty_name(name): name for name in algo_functions.keys()}

    selected_algo_pretty = st.selectbox("🧠 Choisissez l'algorithme :", list(algo_names.keys()))

    if st.button("📊 Lancer la prédiction", type="primary"):
        with st.spinner("🔄 En cours de prédiction..."):
            method_name = algo_names[selected_algo_pretty]
            predict_func = algo_functions[method_name]

            try:
                df_pred = predict_func()

                if isinstance(df_pred, pd.DataFrame):
                    st.success("✅ Prédictions générées avec succès !")

                    st.markdown("### 🔢 Tableau des prédictions (avril → septembre 2025)")
                    st.dataframe(df_pred.style.format({"inflation_pred": "{:.2f}"}), use_container_width=True)

                    # Si tu as une fonction de tracé dans prediction.py, tu peux l'appeler ici
                    if hasattr(module, "plot_predictions"):
                        module.plot_predictions(df_pred)
                else:
                    st.warning("❌ La fonction n'a pas retourné un DataFrame.")
            except Exception as e:
                st.error(f"🚨 Erreur pendant l'exécution du modèle : {e}")
